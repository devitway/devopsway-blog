{{- /* Series Navigation with Progress — показывает все посты серии и текущую позицию. $total берётся из frontmatter-параметра seriesTotal (полный план серии), иначе — число опубликованных. Недостающие (ещё не вышедшие/draft) уровни рисуются как ○ "скоро" БЕЗ ссылки — финал не раскрывается заранее. */ -}} {{- with .Params.series -}} {{- $currentSeries := . -}} {{- $currentPage := $ -}} {{- /* Опубликованные посты серии, сортировка по weight */ -}} {{- $seriesPosts := where site.RegularPages ".Params.series" $currentSeries -}} {{- $seriesPosts = sort $seriesPosts ".Params.weight" "asc" -}} {{- if gt (len $seriesPosts) 1 -}} {{- $published := len $seriesPosts -}} {{- /* Полный размер серии: seriesTotal (если задан и больше опубликованных), иначе — опубликованные */ -}} {{- $total := $published -}} {{- with $currentPage.Params.seriesTotal -}} {{- if gt (int .) $published -}}{{- $total = int . -}}{{- end -}} {{- end -}} {{- /* Позиция текущего поста */ -}} {{- $currentIdx := 0 -}} {{- range $i, $p := $seriesPosts -}} {{- if eq $p.Permalink $currentPage.Permalink -}} {{- $currentIdx = add $i 1 -}} {{- end -}} {{- end -}} {{- $seriesTitle := $currentPage.Params.seriesTitle | default (printf "Серия: %s" $currentSeries) -}}
{{ $seriesTitle }} {{ $currentIdx }}/{{ $total }}
    {{- range $i, $p := $seriesPosts -}} {{- $isCurrent := eq $p.Permalink $currentPage.Permalink -}} {{- $isPast := lt (add $i 1) $currentIdx -}}
  1. {{- if $isCurrent -}} ▶ {{ $p.Title }} {{- else -}} {{ if $isPast }}✓{{ else }}○{{ end }} {{ $p.Title }} {{- end -}}
    2. {{- end -}} {{- /* Ещё не опубликованные уровни (финал придержан): ○ "скоро", без ссылки */ -}} {{- if gt $total $published -}} {{- range $j := (seq (add $published 1) $total) -}}
  2. ○ Уровень {{ sub $j 1 }} – скоро
    3. {{- end -}} {{- end -}}
{{- end -}} {{- end -}}
